function toggleList(yearId, mediaId) { const yearEl = document.getElementById(yearId); const mediaEl = document.getElementById(mediaId); const active = yearEl.classList.contains('expand-list'); resetAll(); if (!active) { yearEl.classList.add('expand-list'); mediaEl.classList.add('hide-pair'); // Dynamic Line Break injection for Mobile if (window.innerWidth < 768) { const breaker = document.createElement('div'); breaker.className = 'row-breaker'; breaker.style.cssText = 'flex-basis: 100%; height: 0; margin: 0; padding: 0;'; mediaEl.parentNode.insertBefore(breaker, mediaEl.nextSibling); } } } function toggleMedia(mediaId, yearId, primaryWidth) { const mediaEl = document.getElementById(mediaId); const yearEl = document.getElementById(yearId); const active = mediaEl.classList.contains('expand-media'); const isMobile = window.innerWidth < 768; resetAll(); if (!active) { mediaEl.classList.add('expand-media'); if (isMobile) { yearEl.classList.add('hide-pair'); // Dynamic Line Break injection for Mobile const breaker = document.createElement('div'); breaker.className = 'row-breaker'; breaker.style.cssText = 'flex-basis: 100%; height: 0; margin: 0; padding: 0;'; mediaEl.parentNode.insertBefore(breaker, mediaEl.nextSibling); } else if (primaryWidth < 50) { yearEl.classList.add('hide-pair'); } else { yearEl.style.flexBasis = "20%"; yearEl.style.maxWidth = "20%"; } } } function resetAll() { // Clear out any old breakdown walls document.querySelectorAll('.row-breaker').forEach(el => el.remove()); document.querySelectorAll('.block').forEach(b => { b.classList.remove('expand-list', 'expand-media', 'hide-pair'); b.style.flexBasis = ""; b.style.maxWidth = ""; }); }
